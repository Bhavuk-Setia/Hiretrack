// ── PAGINATION STATE ───────────────────────────────────────────── let PAGINATION = { page: 1, pageSize: 2000, total: 0, totalPages: 1, loading: false, cancelled: false }; // ── LOAD LEADS (paginated, with retry + timeout) ────────────────── async function loadLeads(page=1, append=false) { if (PAGINATION.loading && !append) return; PAGINATION.loading = true; PAGINATION.cancelled = false; if (!append) { showLoader('Connecting to database…'); LEADS = []; } const params = appRole === 'manager' ? { owner: 'ALL', page, pageSize: PAGINATION.pageSize } : { owner: SESSION.name, page, pageSize: PAGINATION.pageSize }; // Retry up to 3 times before giving up on this page let res = null; for (let attempt = 1; attempt <= 3; attempt++) { res = await apiWithTimeout('getLeads', params, 30000); if (res.ok) break; if (attempt < 3) { await sleep(2000 * attempt); // back off: 2s, 4s } } if (!res || !res.ok) { PAGINATION.loading = false; toast('Failed to load leads. Please refresh.', true); updateLoadingBanner(false); return; } if (append) { LEADS = LEADS.concat(res.leads); } else { LEADS = res.leads; } PAGINATION.page = res.page; PAGINATION.total = res.total; PAGINATION.totalPages = res.totalPages; rebuildAssignDropdown(); updateLeadCount(); if (!append) { // First page loaded — render immediately so user can work render(); } if (res.page < res.totalPages && !PAGINATION.cancelled) { // Show progress banner updateLoadingBanner(true, res.page, res.totalPages, LEADS.length, res.total); PAGINATION.loading = false; await sleep(1500); // wait 1.5s between pages — avoids Apps Script rate limit loadMoreInBackground(res.page + 1); } else { PAGINATION.loading = false; updateLoadingBanner(false); toast('All ' + LEADS.length + ' leads loaded'); } } async function loadMoreInBackground(startPage) { for (let p = startPage; p <= PAGINATION.totalPages; p++) { if (PAGINATION.cancelled) break; PAGINATION.loading = true; const params = appRole === 'manager' ? { owner: 'ALL', page: p, pageSize: PAGINATION.pageSize } : { owner: SESSION.name, page: p, pageSize: PAGINATION.pageSize }; let res = null; for (let attempt = 1; attempt <= 3; attempt++) { res = await apiWithTimeout('getLeads', params, 30000); if (res && res.ok) break; if (attempt < 3) await sleep(3000 * attempt); } if (!res || !res.ok) { // Failed after retries — stop background loading, show resume button PAGINATION.loading = false; updateLoadingBanner(false, p); showResumeButton(p); return; } LEADS = LEADS.concat(res.leads); PAGINATION.page = res.page; rebuildAssignDropdown(); updateLeadCount(); updateLoadingBanner(true, res.page, res.totalPages, LEADS.length, res.total); PAGINATION.loading = false; if (p < PAGINATION.totalPages && !PAGINATION.cancelled) { await sleep(1500); // pause between pages } } if (!PAGINATION.cancelled) { updateLoadingBanner(false); updateLeadCount(); toast('All ' + LEADS.length + ' leads loaded ✓'); } } // API call with a hard timeout so it never hangs forever async function apiWithTimeout(action, body={}, timeoutMs=30000) { const controller = new AbortController(); const timer = setTimeout(() => controller.abort(), timeoutMs); try { const res = await fetch(SCRIPT_URL, { method: 'POST', body: JSON.stringify({ action, ...body }), signal: controller.signal }); clearTimeout(timer); return await res.json(); } catch(e) { clearTimeout(timer); return { ok: false, error: e.name === 'AbortError' ? 'Request timed out' : e.message }; } } async function refreshLeads() { PAGINATION.cancelled = true; // stop any background loading PAGINATION.loading = false; await sleep(200); // let in-flight calls settle PAGINATION.cancelled = false; PAGINATION.page = 1; LEADS = []; await loadLeads(1, false); } // ── PROGRESS BANNER ─────────────────────────────────────────────── function updateLoadingBanner(show, curPage=0, totalPages=0, loaded=0, total=0) { let banner = document.getElementById('load-banner'); if (!show) { if (banner) banner.remove(); return; } if (!banner) { banner = document.createElement('div'); banner.id = 'load-banner'; banner.style.cssText = 'position:fixed;bottom:20px;left:50%;transform:translateX(-50%);background:#1a1a2e;color:#fff;padding:10px 20px;border-radius:10px;font-size:12px;z-index:998;display:flex;align-items:center;gap:12px;box-shadow:0 4px 20px rgba(0,0,0,.3)'; document.body.appendChild(banner); } const pct = total ? Math.round(loaded/total*100) : 0; banner.innerHTML = `
Loading ${loaded.toLocaleString()} of ${total.toLocaleString()} leads (${pct}%) `; } function showResumeButton(fromPage) { let banner = document.getElementById('load-banner'); if (!banner) { banner = document.createElement('div'); banner.id = 'load-banner'; banner.style.cssText = 'position:fixed;bottom:20px;left:50%;transform:translateX(-50%);background:#f59e0b;color:#1a1a2e;padding:10px 20px;border-radius:10px;font-size:12px;z-index:998;display:flex;align-items:center;gap:12px;box-shadow:0 4px 20px rgba(0,0,0,.2)'; document.body.appendChild(banner); } banner.innerHTML = ` ⚠ Loading paused at ${LEADS.length.toLocaleString()} leads `; } function updateLeadCount() { const badges = document.querySelectorAll('.nb-b'); badges.forEach(b => b.textContent = LEADS.length); const rc = document.getElementById('res-count'); if (rc && _curLeads.length) { rc.textContent = applyFilters(_curLeads).length + ' of ' + _curLeads.length + ' shown'; } } function sleep(ms) { return new Promise(r => setTimeout(r, ms)); } function rebuildAssignDropdown() { const sel = document.getElementById('asg-rec'); if (!sel) return; while (sel.options.length > 3) sel.remove(3); RECRUITERS = [...new Set(LEADS.map(l => l.owner).filter(Boolean))]; RECRUITERS.forEach(r => { const o = document.createElement('option'); o.value = r; o.text = r; sel.add(o, 2); }); }
